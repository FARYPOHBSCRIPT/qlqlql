local function loadWindUI()
    local lastError = "unknown"

    for _, url in ipairs(WINDUI_URLS) do
        local ok, result = pcall(function()
            local source = game:HttpGet(url)
            local chunk = loadstring(source)
            if type(chunk) ~= "function" then
                error("loadstring 失败")
            end
            return chunk()
        end)

        if ok and typeof(result) == "table" and type(result.CreateWindow) == "function" then
            return result
        end

        lastError = tostring(result)
        warnf("WindUI 载入失败: %s", lastError)
        task.wait(0.4)
    end

    return nil, lastError
end

local WindUI, loadError = loadWindUI()
if not WindUI then
    showBootError("WindUI 载入失败，请检查执行器的 HttpGet 是否可用。\n" .. tostring(loadError))
    return
end

local CONFIG_FILE = "TURNopensource_config.json"

local canSave = type(writefile) == "function" and type(readfile) == "function"

local function encodeConfig()
    local data = {}
    for key, value in pairs(Config) do
        if typeof(value) == "Color3" then
            data[key] = { __color = true, r = value.R, g = value.G, b = value.B }
        elseif type(value) == "number" or type(value) == "string" or type(value) == "boolean" then
            data[key] = value
        end
    end
    return data
end

local function decodeConfig(data)
    for key, value in pairs(data) do
        local current = Config[key]
        if current ~= nil then
            if typeof(current) == "Color3" and type(value) == "table" and value.__color then
                Config[key] = Color3.new(
                    math.clamp(tonumber(value.r) or 1, 0, 1),
                    math.clamp(tonumber(value.g) or 1, 0, 1),
                    math.clamp(tonumber(value.b) or 1, 0, 1)
                )
            elseif type(value) == type(current) then
                Config[key] = value
            end
        end
    end
end

if canSave then
    local function readConfig(path)
        local ok, raw = pcall(readfile, path)
        if not ok or type(raw) ~= "string" then return nil end
        local decoded, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if decoded and type(data) == "table" then return data end
        return nil
    end

    local data = readConfig(CONFIG_FILE) or readConfig(LEGACY_CONFIG_FILE)
    if data then
        decodeConfig(data)
    end
end

local Range = {}
Range.__index = Range

local RING_SEGMENTS = 56

function Range.new(target)
    local self = setmetatable({}, Range)

    self.Target = target
    self.Radius = 0
    self.Visible = false
    self.Armed = false
    self.LastCFrame = nil

    local model = Instance.new("Model")
    model.Name = "TURN_v1_RangeCircle"
    self.Model = model

    -- 玻璃填充：一片极薄的白色圆盘
    local disc = Instance.new("Part")
    disc.Name = "GlassFill"
    disc.Shape = Enum.PartType.Cylinder
    disc.Anchored = true
    disc.Locked = true
    disc.CanCollide = false
    disc.CanQuery = false
    disc.CanTouch = false
    disc.CastShadow = false
    disc.Material = Enum.Material.SmoothPlastic
    disc.Color = Color3.fromRGB(255, 255, 255)
    disc.Transparency = Config.CircleTransparency
    disc.TopSurface = Enum.SurfaceType.Smooth
    disc.BottomSurface = Enum.SurfaceType.Smooth
    -- Roblox 的 Cylinder 沿 Y 轴延伸，X/Z 是直径；不加旋转就是平放圆盘
    disc.Size = Vector3.new(2, 0.05, 2)
    disc.CFrame = CFrame.new()
    disc.Parent = model
    self.Disc = disc
    self.DiscBase = disc.CFrame

    -- 玻璃边缘：一圈白色细条，拼成完整的圆
    self.Rim = {}
    for i = 1, RING_SEGMENTS do
        local bar = Instance.new("Part")
        bar.Name = "Rim" .. i
        bar.Anchored = true
        bar.Locked = true
        bar.CanCollide = false
        bar.CanQuery = false
        bar.CanTouch = false
        bar.CastShadow = false
        bar.Material = Enum.Material.SmoothPlastic
        bar.Color = Color3.fromRGB(255, 255, 255)
        bar.Transparency = 0.12
        bar.TopSurface = Enum.SurfaceType.Smooth
        bar.BottomSurface = Enum.SurfaceType.Smooth
        bar.Size = Vector3.new(1, 0.06, 0.18)
        bar.CFrame = CFrame.new()
        bar.Parent = model
        self.Rim[i] = bar
        self.RimBase = self.RimBase or {}
        self.RimBase[i] = bar.CFrame
    end

    self.RayParams = RaycastParams.new()
    self.RayParams.FilterType = Enum.RaycastFilterType.Exclude
    self.RayParams.FilterDescendantsInstances = {}
    self.RayParams.IgnoreWater = true

    model.Parent = nil

    self:SetRadius(Config.Radius)
    return self
end

function Range:SetRadius(radius)
    radius = math.max(1, tonumber(radius) or Config.Radius)
    if math.abs(radius - self.Radius) < 0.01 then return end

    self.Radius = radius
    local diameter = radius * 2

    self.Disc.Size = Vector3.new(diameter, 0.05, diameter)

    local step = (math.pi * 2) / RING_SEGMENTS
    local length = 2 * radius * math.sin(step / 2) * 1.2

    for i, bar in ipairs(self.Rim) do
        local phi = (i - 1) * step
        bar.Size = Vector3.new(length, 0.06, 0.18)
        bar.CFrame = CFrame.new(math.cos(phi) * radius, 0, math.sin(phi) * radius)
            * CFrame.Angles(0, -phi - math.rad(90), 0)
        self.RimBase[i] = bar.CFrame
    end

    self.DiscBase = self.Disc.CFrame

    -- 半径变化时重建了局部坐标，立刻放回杀手脚下，避免闪到世界原点
    if self.LastCFrame then
        self:ApplyCFrame(self.LastCFrame)
    end
end

-- 直接写每个零件的 CFrame，避免 Model 枢轴带着圆盘旋转，导致圈立起来
function Range:ApplyCFrame(cf)
    if not self.Model then return end
    self.Disc.CFrame = cf * self.DiscBase
    for i, bar in ipairs(self.Rim) do
        bar.CFrame = cf * self.RimBase[i]
    end
end

function Range:SetVisible(visible)
    self.Visible = visible
    if not visible and self.Model.Parent then
        self.Model.Parent = nil
    end
end

function Range:SetArmed(armed)
    self.Armed = armed
    local transparency = armed and 0.02 or 0.12
    for _, bar in ipairs(self.Rim) do
        bar.Transparency = transparency
    end
end

function Range:Flash()
    if not self.Model then return end
    for _, bar in ipairs(self.Rim) do
        bar.Transparency = 0
    end

    task.delay(0.12, function()
        if not self.Model then return end
        local transparency = self.Armed and 0.02 or 0.12
        for _, bar in ipairs(self.Rim) do
            bar.Transparency = transparency
        end
    end)
end

function Range:Update(dt)
    if not self.Visible then
        if self.Model.Parent then self.Model.Parent = nil end
        return
    end

    local target = self.Target
    local root = target and target.Parent and target:FindFirstChild("HumanoidRootPart")
    local humanoid = target and target:FindFirstChildOfClass("Humanoid")
    if not root or (humanoid and humanoid.Health <= 0) then
        if self.Model.Parent then self.Model.Parent = nil end
        return
    end

    local origin = root.Position

    self.RayParams.FilterDescendantsInstances = { target }
    local hit = Workspace:Raycast(origin, Vector3.new(0, -14, 0), self.RayParams)
    local groundY = hit and (hit.Position.Y + 0.06) or (origin.Y - 2.7)

    -- 0 帧：直接贴到杀手脚下，不做插值
    local cf = CFrame.new(origin.X, groundY, origin.Z)
    self.LastCFrame = cf
    self:ApplyCFrame(cf)

    if self.Model.Parent ~= Workspace then
        self.Model.Parent = Workspace
    end

    local pulse = 0
    if Config.Pulse then
        local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local near = false
        if myRoot then
            local dx = myRoot.Position.X - origin.X
            local dz = myRoot.Position.Z - origin.Z
            near = math.sqrt(dx * dx + dz * dz) <= Config.Radius
        end
        local speed = (Config.AutoBlock and near) and 4.2 or 2.0
        pulse = math.sin(os.clock() * speed) * 0.035
    end
    self.Disc.Transparency = math.clamp(Config.CircleTransparency + pulse, 0.35, 0.98)
end

function Range:Destroy()
    if self.Model then
        self.Model:Destroy()
        self.Model = nil
    end
    self.Target = nil
end

-- 「最后记忆」的杀手角色被放进 workspace.Players.Killers；
-- 找不到这个分组时，退化为「所有其他玩家」，方便其它场景复用。
local function getCandidates()
    local folder = getKillerFolder()
    local list = {}

    if folder then
        local seen = {}
        local function add(model)
            if model:IsA("Model") and not seen[model] then
                seen[model] = true
                list[#list + 1] = {
                    Model = model,
                    Player = Players:GetPlayerFromCharacter(model),
                }
            end
        end

        for _, child in ipairs(folder:GetChildren()) do
            add(child)
        end
        for _, child in ipairs(folder:GetDescendants()) do
            add(child)
        end
    else
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                list[#list + 1] = { Model = player.Character, Player = player }
            end
        end
    end

    return list
end

local function getTargetsInRange(radius)
    local result = {}

    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return result end

    local origin = root.Position

    for _, candidate in ipairs(getCandidates()) do
        local model = candidate.Model

        if model and model ~= character and model.Parent then
            local otherRoot = model:FindFirstChild("HumanoidRootPart")
                or model:FindFirstChild("QueryHitbox")
                or model:FindFirstChild("Torso")
            local humanoid = model:FindFirstChildOfClass("Humanoid")

            if otherRoot and humanoid and humanoid.Health > 0 then
                local dx = otherRoot.Position.X - origin.X
                local dz = otherRoot.Position.Z - origin.Z
                local distance = math.sqrt(dx * dx + dz * dz)

                if distance <= radius then
                    table.insert(result, {
                        Player   = candidate.Player,
                        Model    = model,
                        Name     = candidate.Player and candidate.Player.Name or model.Name,
                        Distance = distance,
                    })
                end
            end
        end
    end

    table.sort(result, function(a, b)
        return a.Distance < b.Distance
    end)

    return result
end

local rings = {}

local function syncRings()
    if destroyed then return end

    local wanted = {}
    if Config.ShowCircle then
        for _, candidate in ipairs(getCandidates()) do
            local model = candidate.Model
            if model and model.Parent and model ~= LocalPlayer.Character then
                local humanoid = model:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    wanted[model] = true
                end
            end
        end
    end

    for model in pairs(wanted) do
        if not rings[model] then
            local ring = Range.new(model)
            ring:SetRadius(Config.Radius)
            ring:SetArmed(Config.AutoBlock)
            ring:SetVisible(true)
            rings[model] = ring
        end
    end

    for model, ring in pairs(rings) do
        if not wanted[model] then
            ring:Destroy()
            rings[model] = nil
        end
    end
end

local function setRingsVisible(visible)
    Config.ShowCircle = visible
    for _, ring in pairs(rings) do
        ring:SetVisible(visible)
    end
    if visible then syncRings() end
end

local function setRingsArmed(armed)
    for _, ring in pairs(rings) do
        ring:SetArmed(armed)
    end
end

local function setRingsRadius(radius)
    for _, ring in pairs(rings) do
        ring:SetRadius(radius)
    end
end

        local ok3, f3 = pcall(function() return getgenv().firesignal end)
        if ok3 then FireSignal = pick(f3) end
    end
end

local function isRemoteInstance(instance)
    if not instance then return false end
    local ok, result = pcall(function()
        return instance:IsA("RemoteEvent") or instance:IsA("UnreliableRemoteEvent")
    end)
    return ok and result or false
end

local REMOTE_PATHS = {
    { "Modules", "Network", "Network", "RemoteEvent" },
    { "Modules", "Network", "RemoteEvent" },
}

local blockRemote = nil

local function tryRemotePath(path)
    local ok, node = pcall(function()
        local current = ReplicatedStorage
        for _, name in ipairs(path) do
            current = current:FindFirstChild(name)
            if not current then return nil end
        end
        return current
    end)

    if ok then
        if isRemoteInstance(node) then return node end
        if node then
            local child = node:FindFirstChildOfClass("RemoteEvent")
            if isRemoteInstance(child) then return child end
        end
    end
    return nil
end

local function ensureBlockRemote()
    if blockRemote and blockRemote.Parent then return blockRemote end
    blockRemote = nil

    -- 优先找 Modules/Network 下真正的 RemoteEvent
    local okDeep, deepRemote = pcall(function()
        local modules = ReplicatedStorage:FindFirstChild("Modules")
        if not modules then return nil end
        local network = modules:FindFirstChild("Network", true)
        if not network then return nil end
        local inner = network:FindFirstChild("Network")
        local target = inner or network
        return target:FindFirstChildOfClass("RemoteEvent")
            or network:FindFirstChildOfClass("RemoteEvent")
    end)

    if okDeep and isRemoteInstance(deepRemote) then
        blockRemote = deepRemote
        return deepRemote
    end

    for _, path in ipairs(REMOTE_PATHS) do
        local remote = tryRemotePath(path)
        if remote then
            blockRemote = remote
            return remote
        end
    end
    return nil
end

                if number and number > 0.05 then

local function localRoot()
    local character = LocalPlayer.Character
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function targetRoot(model)
    if not model then return nil end
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("QueryHitbox")
        or model:FindFirstChild("Torso")
end

local function distanceToLocal(model)
    local root = localRoot()
    local other = targetRoot(model)
    if not (root and other) then return nil end

    local dx = other.Position.X - root.Position.X
    local dz = other.Position.Z - root.Position.Z
    return math.sqrt(dx * dx + dz * dz)
end

local function isInCircle(model)
    local distance = distanceToLocal(model)
    return distance ~= nil and distance <= Config.Radius + 0.25
end

local function isLocalKiller()
    local folder = getKillerFolder()
    local character = LocalPlayer.Character
    return folder ~= nil and character ~= nil and character.Parent == folder
end

local function isKillerModel(model)
    if not model or model == LocalPlayer.Character then return false end

    local folder = getKillerFolder()
    if folder then
        if model.Parent == folder or model:IsDescendantOf(folder) then
            return true
        end
    end

    if model:FindFirstAncestor("Killers") then return true end

    local ok, isKiller = pcall(function() return model:GetAttribute("IsKiller") end)
    if ok and isKiller == true then return true end

    if not folder then return true end
    return false
end

local function killerFacesMe(killer)
    if not Config.FaceCheck then return true end

    local myRoot = localRoot()
    local killerRoot = targetRoot(killer)
    if not (myRoot and killerRoot) then return true end

    local look = killerRoot.CFrame.LookVector
    local toMe = myRoot.Position - killerRoot.Position
    local flatLook = Vector3.new(look.X, 0, look.Z)
    local flatTo = Vector3.new(toMe.X, 0, toMe.Z)
    if flatLook.Magnitude < 1e-3 or flatTo.Magnitude < 1e-3 then return true end

    local halfAngle = math.clamp(Config.FaceAngle or 180, 20, 180) / 2
    local minDot = math.cos(math.rad(halfAngle))
    return flatLook.Unit:Dot(flatTo.Unit) >= minDot
end

local function wallBetween(killer)
    if not Config.WallCheck then return false end

    local myCharacter = LocalPlayer.Character
    local myRoot = localRoot()
    local killerRoot = targetRoot(killer)
    if not (myCharacter and myRoot and killerRoot) then return false end

    local ok, blocked = pcall(function()
        local from = myRoot.Position + Vector3.new(0, 1.5, 0)
        local to = killerRoot.Position + Vector3.new(0, 1.5, 0)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { myCharacter, killer }
        params.IgnoreWater = true
        local hit = Workspace:Raycast(from, to - from, params)
        if not hit or not hit.Instance then return false end

        -- 别的角色/杀手不算墙；多杀手贴在一起时不能互相挡住
        local hitModel = hit.Instance:FindFirstAncestorOfClass("Model")
        if hitModel and hitModel:FindFirstChildOfClass("Humanoid") then
            return false
        end

        return hit.Instance.CanCollide == true
    end)

    return ok and blocked == true
end

local function findBlockableKiller(position)
    local best, bestScore = nil, math.huge

    for _, candidate in ipairs(getCandidates()) do
        local model = candidate.Model
        if model and model.Parent and model ~= LocalPlayer.Character then
            local distance = distanceToLocal(model)
            if distance and distance <= Config.Radius + 0.25
                and killerFacesMe(model)
                and not wallBetween(model) then
                local score = distance
                if position then
                    local root = targetRoot(model)
                    if root then
                        score = (root.Position - position).Magnitude
                    end
                end
                if score < bestScore then
                    best, bestScore = model, score
                end
            end
        end
    end

    return best
end

local BLOCK_DEBOUNCE = 0.25
local lastBlockAt = 0

local function triggerBlock(killer, source)
    if destroyed or not Config.AutoBlock then return false end
    if not killer or not killer.Parent then return false end
    if killer == LocalPlayer.Character then return false end
    if isLocalKiller() then return false end
    if not isKillerModel(killer) then return false end

    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return false end

    if not isInCircle(killer) then return false end
    if not killerFacesMe(killer) then return false end
    if wallBetween(killer) then return false end

    local now = os.clock()
    if now - lastBlockAt < BLOCK_DEBOUNCE then return false end
    if Config.BlockReadyCheck and not blockAbilityReady() then return false end
    lastBlockAt = now

    local ok = fireBlockAbility()
    if Config.BlockDebug then
        print(string.format("[TURNopensource] block <- %s | %s | dist=%.1f | ok=%s",
            tostring(killer.Name), tostring(source), distanceToLocal(killer) or -1, tostring(ok)))
    end

    if ok then
        local ring = rings[killer]
        if ring then
            pcall(function() ring:Flash() end)
        end
    end
    return ok
end

local faceCones = {}

local function makeConePart(shape)
    local part = Instance.new("Part")
    part.Name = "TURN_v1_FaceCone"
    part.Anchored = true
    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.CastShadow = false
    part.Material = Enum.Material.Neon
    part.Color = Color3.fromRGB(255, 60, 60)
    part.Transparency = 1
    if shape then part.Shape = shape end
    part.Parent = Workspace
    return part
end

local function createFaceCone()
    local cone = {
        Lines = {},
        Tips = {},
        Params = RaycastParams.new(),
    }
    cone.Params.FilterType = Enum.RaycastFilterType.Exclude
    cone.Params.IgnoreWater = true

    for i = 1, 4 do
        local part = makeConePart(nil)
        part.Size = Vector3.new(0.12, 0.12, 1)
        cone.Lines[i] = part
    end
    for i = 1, 3 do
        local part = makeConePart(Enum.PartType.Ball)
        part.Size = Vector3.new(0.5, 0.5, 0.5)
        cone.Tips[i] = part
    end

    return cone
end

local function setConeLine(part, a, b)
    local distance = (b - a).Magnitude
    if distance < 1e-3 then
        part.Transparency = 1
        return
    end
    part.Size = Vector3.new(0.12, 0.12, distance)
    part.CFrame = CFrame.new((a + b) / 2, b)
    part.Transparency = 0.15
end

local function syncFaceCones()
    if destroyed then return end

    local wanted = {}
    if Config.FaceVisual then
        for _, candidate in ipairs(getCandidates()) do
            local model = candidate.Model
            if model and model.Parent and model ~= LocalPlayer.Character then
                local humanoid = model:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    wanted[model] = true
                end
            end
        end
    end

    for model in pairs(wanted) do
        if not faceCones[model] then
            faceCones[model] = createFaceCone()
        end
    end

    for model, cone in pairs(faceCones) do
        if not wanted[model] then
            for _, part in ipairs(cone.Lines) do part:Destroy() end
            for _, part in ipairs(cone.Tips) do part:Destroy() end
            faceCones[model] = nil
        end
    end
end

local function updateFaceCones()
    if destroyed then return end

    for model, cone in pairs(faceCones) do
        local root = targetRoot(model)
        local humanoid = model:FindFirstChildOfClass("Humanoid")
        if not root or not model.Parent or (humanoid and humanoid.Health <= 0) or not Config.FaceVisual then
            for _, part in ipairs(cone.Lines) do part.Transparency = 1 end
            for _, part in ipairs(cone.Tips) do part.Transparency = 1 end
        else
            local look = root.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z)
            if flat.Magnitude < 1e-3 then
                for _, part in ipairs(cone.Lines) do part.Transparency = 1 end
                for _, part in ipairs(cone.Tips) do part.Transparency = 1 end
            else
                flat = flat.Unit
                cone.Params.FilterDescendantsInstances = { model }
                local hit = Workspace:Raycast(root.Position, Vector3.new(0, -14, 0), cone.Params)
                local groundY = hit and (hit.Position.Y + 0.12) or (root.Position.Y - 2.5)
                local origin = Vector3.new(root.Position.X, groundY, root.Position.Z)

                local half = math.rad(math.clamp(Config.FaceAngle, 20, 180) / 2)
                local right = Vector3.new(-flat.Z, 0, flat.X)
                local leftDir = (flat * math.cos(half) + right * math.sin(half)).Unit
                local rightDir = (flat * math.cos(half) - right * math.sin(half)).Unit
                local distance = Config.FaceDist
                local tip = origin + flat * distance
                local leftPoint = origin + leftDir * distance
                local rightPoint = origin + rightDir * distance

                setConeLine(cone.Lines[1], origin, leftPoint)
                setConeLine(cone.Lines[2], origin, rightPoint)
                setConeLine(cone.Lines[3], leftPoint, tip)
                setConeLine(cone.Lines[4], tip, rightPoint)

                cone.Tips[1].Position = tip
                cone.Tips[2].Position = leftPoint
                cone.Tips[3].Position = rightPoint

                local color = (isInCircle(model) and Config.AutoBlock)
                    and Color3.fromRGB(0, 255, 110)
                    or Color3.fromRGB(255, 60, 60)

                for _, part in ipairs(cone.Lines) do
                    part.Color = color
                    part.Transparency = 0.15
                end
                for _, part in ipairs(cone.Tips) do
                    part.Color = color
                    part.Transparency = 0.2
                end
            end
        end
    end
end

local watchedSounds = setmetatable({}, { __mode = "k" })
local hookedModels = setmetatable({}, { __mode = "k" })
local hookedAnimators = setmetatable({}, { __mode = "k" })

local function soundIdOf(sound)
    if not sound then return nil end
    local ok, raw = pcall(function() return sound.SoundId end)
    if not ok or not raw then return nil end
    return tostring(raw):match("%d+")
end

local function findOwnerModel(instance)
    local node = instance
    for _ = 1, 10 do
        node = node.Parent
        if not node or node == Workspace then return nil end
        if node:IsA("Model") and node:FindFirstChildOfClass("Humanoid") then
            return node
        end
    end
    return nil
end

local function nearestKiller(position)
    local best, bestDistance = nil, math.huge

    for _, candidate in ipairs(getCandidates()) do
        local model = candidate.Model
        if model and model.Parent and model ~= LocalPlayer.Character then
            local root = targetRoot(model)
            if root then
                local distance
                if position then
                    distance = (root.Position - position).Magnitude
                else
                    distance = distanceToLocal(model)
                end
                if distance and distance < bestDistance then
                    best, bestDistance = model, distance
                end
            end
        end
    end

    return best, bestDistance
end

local function resolveKillerForSound(sound)
    local owner = findOwnerModel(sound)

    if owner == LocalPlayer.Character then
        -- 命中音效挂在本地角色身上：从所有圈内杀手里挑一个能格挡的
        return findBlockableKiller(nil)
    end

    if owner then
        -- 音效明确属于某个杀手：只认这个杀手，避免多杀手时串线
        if isKillerModel(owner) then return owner end
        return nil
    end

    local parent = sound.Parent
    if parent and parent:IsA("BasePart") then
        -- 无主音效挂在特效部件上：挑离音效最近、且在圈内/面向/无墙的杀手
        local killer = findBlockableKiller(parent.Position)
        if killer then
            local root = targetRoot(killer)
            if root and (root.Position - parent.Position).Magnitude <= 16 then
                return killer
            end
        end
        return nil
    end

    -- 全局音效：多杀手时不能只看最近，要找一个真正能挡的
    return findBlockableKiller(nil)
end

local function watchSound(sound)
    if watchedSounds[sound] then return end
    if not sound:IsA("Sound") then return end

    local id = soundIdOf(sound)
    if not id or not SOUND_BLOCK_IDS[id] then return end
    watchedSounds[sound] = true

    local function check()
        if destroyed or not Config.AutoBlock or not Config.SoundBlock then return end

        local currentId = soundIdOf(sound)
        if not currentId or not SOUND_BLOCK_IDS[currentId] then return end

        local killer = resolveKillerForSound(sound)
        if killer then
            triggerBlock(killer, "sound:" .. currentId)
        end
    end

    sound.Played:Connect(function()
        pcall(check)
    end)
    sound:GetPropertyChangedSignal("Playing"):Connect(function()
        if sound.Playing then pcall(check) end
    end)
    sound:GetPropertyChangedSignal("SoundId"):Connect(function()
        local currentId = soundIdOf(sound)
        if currentId and SOUND_BLOCK_IDS[currentId] then
            pcall(check)
        end
    end)

    if sound.Playing then check() end
end

local function isAttackAnimation(name)
    local normalized = normalizeName(name)
    if normalized == "" then return false end
    if ATTACK_ANIM_NAMES[normalized] then return true end

    for key in pairs(ATTACK_ANIM_NAMES) do
        if normalized:find(key, 1, true) then return true end
    end
    return false
end

local scannedTrackAt = setmetatable({}, { __mode = "k" })

local function animationNameOf(track)
    local animation = track and track.Animation
    return (animation and animation.Name) or (track and track.Name) or ""
end

local function checkAnimation(killer, track)
    if destroyed or not Config.AutoBlock or not Config.AnimBlock then return end
    if not killer or not killer.Parent or not track then return end

    local name = animationNameOf(track)
    if not isAttackAnimation(name) then return end

    local now = os.clock()
    if now - (scannedTrackAt[track] or 0) < 0.3 then return end
    scannedTrackAt[track] = now

    triggerBlock(killer, "anim:" .. tostring(name))
end

local function scanPlayingAnimations()
    for animator, killer in pairs(hookedAnimators) do
        if killer and killer.Parent and animator.Parent then
            local ok, tracks = pcall(function()
                return animator:GetPlayingAnimationTracks()
            end)

            if ok then
                for _, track in ipairs(tracks) do
                    -- 只兜底“漏掉的起手”：已经由 AnimationPlayed 处理过的不重复扫
                    if not scannedTrackAt[track] then
                        checkAnimation(killer, track)
                    end
                end
            end
        end
    end
end

local function hookAnimator(killer, animator)
    if not animator or hookedAnimators[animator] then return end
    hookedAnimators[animator] = killer

    animator.AnimationPlayed:Connect(function(track)
        pcall(checkAnimation, killer, track)
    end)

    pcall(function()
        for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
            checkAnimation(killer, track)
        end
    end)
end

local function hookKillerModel(model)
    if hookedModels[model] then return end
    hookedModels[model] = true

    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if humanoid then
        hookAnimator(model, humanoid:FindFirstChildOfClass("Animator"))
    end

    for _, descendant in ipairs(model:GetDescendants()) do
        if descendant:IsA("Sound") then
            watchSound(descendant)
        end
    end

    model.DescendantAdded:Connect(function(descendant)
        if destroyed then return end
        pcall(function()
            if descendant:IsA("Sound") then
                watchSound(descendant)
            elseif descendant:IsA("Animator") then
                hookAnimator(model, descendant)
            end
        end)
    end)
end

local watchedKillerFolder = nil

local function ensureKillerFolderHook()
    local folder = getKillerFolder()
    if folder and folder ~= watchedKillerFolder then
        watchedKillerFolder = folder
        folder.ChildAdded:Connect(function(model)
            if destroyed then return end
            pcall(function()
                local humanoid = model:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    hookKillerModel(model)
                end
                syncRings()
                syncFaceCones()
            end)
        end)
    end
end

local function scanDetection()
    if destroyed then return end

    ensureKillerFolderHook()

    for _, candidate in ipairs(getCandidates()) do
        local model = candidate.Model
        if model and model.Parent then
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            if not humanoid or humanoid.Health > 0 then
                hookKillerModel(model)
            end
        end
    end

    scanPlayingAnimations()
    syncRings()
    syncFaceCones()
end

local function espColors(color)
    local h, s, v = color:ToHSV()
    return Color3.fromHSV(h, s, math.clamp(v * 0.8, 0, 1)),
        Color3.fromHSV(h, s, math.clamp(v * 1.1, 0, 1))

local function destroyESPRecord(record)
    if not record then return end
    if record.Highlight then
        pcall(function() record.Highlight:Destroy() end)

    if record.Billboard then
        pcall(function() record.Billboard:Destroy() end)

local function espRootOf(instance)
    if not instance then return nil end

    if instance:IsA("Model") then
        return instance:FindFirstChild("HumanoidRootPart")
            or instance:FindFirstChild("Head")
            or instance.PrimaryPart
            or instance:FindFirstChildWhichIsA("BasePart", true)
    elseif instance:IsA("BasePart") then
        return instance
    elseif instance:IsA("Tool") then
        return instance:FindFirstChild("Handle")
            or instance:FindFirstChildWhichIsA("BasePart", true)

    return instance:FindFirstChildWhichIsA("BasePart", true)
end

local function updateESPRecord(record)
    local instance = record.Instance
    if not instance or not instance.Parent then return false end

    local root = espRootOf(instance)
    if not root then return false end

    local humanoid = instance:IsA("Model") and instance:FindFirstChildOfClass("Humanoid") or nil
    if humanoid and humanoid.Health <= 0 then return false end

    if not record.Highlight or not record.Highlight.Parent then return false end
    if not record.Billboard or not record.Billboard.Parent then return false end

    local camera = Workspace.CurrentCamera
    if not camera then return true end

    local distance = (camera.CFrame.Position - root.Position).Magnitude
    local maxDistance = math.max(Config.ESPMaxDistance, 30)

    if distance > maxDistance then
        record.Highlight.Enabled = false
        record.Billboard.Enabled = false
        return true
    end

    local normalized = math.clamp((distance - 8) / math.max(maxDistance - 8, 1), 0, 1)
    local fill, outline = espColors(record.Color)

    record.Highlight.Enabled = true
    record.Highlight.FillColor = fill
    record.Highlight.OutlineColor = outline
    record.Highlight.FillTransparency = 0.42 + normalized * 0.5
    record.Highlight.OutlineTransparency = normalized * 0.85

    record.Billboard.Enabled = Config.ESPShowName
    record.Billboard.MaxDistance = maxDistance + 50
    record.Label.TextColor3 = outline
    record.Label.TextTransparency = normalized * 0.55

local function updateESP()
    if destroyed then return end

    local records = ESP.Records
    local targets = {}

    if Config.ESP then
        addHumanoids(targets, getKillerFolder(), Config.ESPKillers, Config.ESPColorKiller)
        addHumanoids(targets, getSurvivorsFolder(), Config.ESPSurvivors, Config.ESPColorSurvivor)
        addGlobalHumanoids(targets)
        addGenerators(targets)
        addItems(targets)
        addOther(targets)
    end

    for instance, info in pairs(targets) do
        local record = records[instance]
        if not record then
            record = createESPRecord(instance, info.Color, info.TextFunc)
            records[instance] = record
        end

        if record then
            record.Color = info.Color
            record.TextFunc = info.TextFunc
            if not updateESPRecord(record) then
                destroyESPRecord(record)
                records[instance] = nil
            end
        end
    end

    for instance, record in pairs(records) do
        if not targets[instance] then
            destroyESPRecord(record)
            records[instance] = nil
        end
    end
end

local function clearESP()
    for instance, record in pairs(ESP.Records) do
        destroyESPRecord(record)
        ESP.Records[instance] = nil
    end
end

local aimTarget = nil
local aimEndTime = 0
local aimPrevCamType = nil
local aimPrevAutoRotate = nil

local function getClosestKillerInRange(maxDistance)
    local myCharacter = LocalPlayer.Character
    local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local best, bestDistance = nil, maxDistance or 30

    for _, candidate in ipairs(getCandidates()) do
        local model = candidate.Model
        if model and model.Parent and model ~= myCharacter then
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            local root = targetRoot(model)
            if humanoid and humanoid.Health > 0 and root then
                local distance = (root.Position - myRoot.Position).Magnitude
                if distance <= bestDistance then
                    best, bestDistance = model, distance
                end
            end
        end
    end

    return best
end

local function restoreAimState()
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid and aimPrevAutoRotate ~= nil then
        pcall(function() humanoid.AutoRotate = true end)
    end
    aimPrevAutoRotate = nil

    local camera = Workspace.CurrentCamera
    if camera and aimPrevCamType ~= nil and camera.CameraType == Enum.CameraType.Scriptable then
        pcall(function() camera.CameraType = aimPrevCamType end)
    end
    aimPrevCamType = nil
end

local function faceNow(killer)
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not (root and humanoid) then return end

    local killerRoot = targetRoot(killer)
    if not killerRoot then return end

    if aimPrevAutoRotate == nil then
        aimPrevAutoRotate = humanoid.AutoRotate
    end
    if humanoid.AutoRotate then
        pcall(function() humanoid.AutoRotate = false end)
    end

    local velocity = killerRoot.AssemblyLinearVelocity
    if typeof(velocity) ~= "Vector3" then velocity = Vector3.new() end
    local aimPosition = killerRoot.Position + velocity * 0.1
    local myPosition = root.Position
    local flat = Vector3.new(aimPosition.X, myPosition.Y, aimPosition.Z)

    if (flat - myPosition).Magnitude > 0.05 then
        root.CFrame = CFrame.lookAt(myPosition, flat)
    end
end

local function startAim(killer, duration)
    if not killer or not killer.Parent then return end
    aimTarget = killer
    aimEndTime = os.clock() + (duration or Config.PunchAimDuration or 0.75)
    faceNow(killer)
end

local AIM_LEAD = 0.1

local function punchAimTick()
    if not Config.PunchAim then
        if aimPrevAutoRotate ~= nil or aimPrevCamType ~= nil then
            restoreAimState()
        end
        return
    end

    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not (root and humanoid) or os.clock() > aimEndTime then
        if aimPrevAutoRotate ~= nil or aimPrevCamType ~= nil then
            restoreAimState()
        end
        return
    end

    local target = aimTarget
    if not target or not target.Parent then
        if aimPrevAutoRotate ~= nil or aimPrevCamType ~= nil then
            restoreAimState()
        end
        return
    end

    local killerRoot = targetRoot(target)
    if not killerRoot then return end

    if aimPrevAutoRotate == nil then
        aimPrevAutoRotate = humanoid.AutoRotate
    end
    if humanoid.AutoRotate then
        pcall(function() humanoid.AutoRotate = false end)
    end

    local velocity = killerRoot.AssemblyLinearVelocity
    if typeof(velocity) ~= "Vector3" then velocity = Vector3.new() end
    local aimPosition = killerRoot.Position + velocity * AIM_LEAD
    local myPosition = root.Position
    local flat = Vector3.new(aimPosition.X, myPosition.Y, aimPosition.Z)
    local distance = (flat - myPosition).Magnitude

    if distance > 0.05 then
        local want = CFrame.lookAt(myPosition, flat)
        if distance < 8 then
            root.CFrame = want
        else
            root.CFrame = root.CFrame:Lerp(want, 0.65)
        end
    end

    local camera = Workspace.CurrentCamera
    if camera and Config.AimCamLock then
        if aimPrevCamType == nil then
            aimPrevCamType = camera.CameraType
        end
        if camera.CameraType ~= Enum.CameraType.Scriptable then
            pcall(function() camera.CameraType = Enum.CameraType.Scriptable end)
        end
        local direction = flat - myPosition
        if direction.Magnitude > 0.1 then
            local unit = Vector3.new(direction.X, 0, direction.Z).Unit
            local currentOffset = camera.CFrame.Position - myPosition
            local camDistance = math.clamp(currentOffset.Magnitude, 6, 30)
            local camHeight = math.clamp(currentOffset.Y, 1, 8)
            local camPosition = myPosition + Vector3.new(0, camHeight + 1.5, 0) - unit * camDistance
            local lookPoint = killerRoot.Position + Vector3.new(0, 1.5, 0)
            camera.CFrame = camera.CFrame:Lerp(CFrame.lookAt(camPosition, lookPoint), 0.6)
        end
    elseif camera and aimPrevCamType ~= nil then
        pcall(function() camera.CameraType = aimPrevCamType end)
        aimPrevCamType = nil
    end
end

local function aimOnce()
    if not Config.PunchAim then return end
    local killer = getClosestKillerInRange(Config.PunchAimRange)
    if killer then
        startAim(killer, Config.PunchAimDuration)
    end
end

local hdtActive = false
local hdtTarget = nil
local hdtUntil = 0
local hdtConn = nil
local hdtBv = nil
local hdtSaved = nil

local function stopHDT()
    if hdtConn then
        pcall(function() hdtConn:Disconnect() end)
        hdtConn = nil
    end
    if hdtBv then
        pcall(function() hdtBv:Destroy() end)
        hdtBv = nil
    end
    if hdtSaved then
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            pcall(function()
                humanoid.WalkSpeed = hdtSaved.walk
                humanoid.JumpPower = hdtSaved.jump
                humanoid.AutoRotate = true
            end)
        end
        hdtSaved = nil
    end
    hdtActive = false
    hdtTarget = nil
end

local function beginHDT(target)
    if hdtActive or not target or not target.Parent then return end

    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local targetPart = target:FindFirstChild("HumanoidRootPart")
        or target:FindFirstChild("QueryHitbox")
    if not (root and humanoid and targetPart) then return end

    hdtActive = true
    hdtTarget = target

    local duration = Config.SmartBlockHold or 1.5
    hdtUntil = os.clock() + duration
    hdtSaved = {
        walk = humanoid.WalkSpeed,
        jump = humanoid.JumpPower,
    }

    humanoid.WalkSpeed = 0
    humanoid.JumpPower = 0
    humanoid.AutoRotate = false

    local bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    bodyVelocity.Velocity = Vector3.new()
    bodyVelocity.Parent = root
    hdtBv = bodyVelocity

    local startY = root.Position.Y
    local started = os.clock()

    hdtConn = RunService.RenderStepped:Connect(function()
        if not hdtActive then return end

        local ok = pcall(function()
            local currentCharacter = LocalPlayer.Character
            local currentRoot = currentCharacter and currentCharacter:FindFirstChild("HumanoidRootPart")
            if not currentRoot then stopHDT() return end

            local currentTarget = target.Parent
                and (target:FindFirstChild("HumanoidRootPart") or target:FindFirstChild("QueryHitbox"))
            if not currentTarget then stopHDT() return end
            if os.clock() > hdtUntil then stopHDT() return end

            local myPosition = currentRoot.Position
            local targetPosition = currentTarget.Position
            local toTarget = Vector3.new(
                targetPosition.X - myPosition.X,
                0,
                targetPosition.Z - myPosition.Z
            )
            local distance = toTarget.Magnitude
            if distance <= 1.5 then
                stopHDT()
                return
            end

            local direction = toTarget.Unit
            local baseSpeed = math.max(tonumber(Config.SmartBlockDash) or 8, 6)
            local elapsed = os.clock() - started
            local speed = baseSpeed * (0.5 + 0.5 * math.clamp(elapsed / 0.15, 0, 1))
            if distance < 8 then
                speed = speed * math.clamp((distance - 1.5) / 6.5, 0.3, 1)
            end

            if hdtBv and hdtBv.Parent then
                hdtBv.Velocity = Vector3.new(direction.X * speed, 0, direction.Z * speed)
            end

            currentRoot.CFrame = CFrame.lookAt(myPosition,
                Vector3.new(targetPosition.X, myPosition.Y, targetPosition.Z))

            local verticalDistance = math.abs(currentRoot.Position.Y - startY)
            if verticalDistance > 1.5 then
                local rotation = currentRoot.CFrame - currentRoot.CFrame.Position
                currentRoot.CFrame = CFrame.new(currentRoot.Position.X, startY, currentRoot.Position.Z)
                    * rotation
            end
        end)

        if not ok then stopHDT() end
    end)

    task.delay(duration + 0.2, function()
        if hdtActive then stopHDT() end
    end)
end

-- 实时监测本地角色正在播放的动画（不再依赖按钮/有效格挡）
local hdtTrackAt = setmetatable({}, { __mode = "k" })

local function scanHDTAnimations()
    if destroyed or not Config.SmartBlock or hdtActive then return end

    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
    if not animator then return end

    local ok, tracks = pcall(function()
        return animator:GetPlayingAnimationTracks()
    end)
    if not ok or type(tracks) ~= "table" then return end

    local now = os.clock()

    for _, track in ipairs(tracks) do
        local animation = track.Animation
        local id, name
        pcall(function()
            id = animation and tostring(animation.AnimationId):match("%d+")
            name = animation and tostring(animation.Name) or ""
        end)

        if isHDTBlockAnim(id) then
            local last = hdtTrackAt[track]
            if not last or now - last >= 2 then
                hdtTrackAt[track] = now
                local killer = nearestHDTKiller(Config.SmartBlockRange)
                if killer then
                    beginHDT(killer)
                end
                break
            end
        end
    end
end

local hdtScanConn = RunService.RenderStepped:Connect(function()
    if destroyed then return end
    pcall(scanHDTAnimations)
end)

local function setFPSPing(value)
    if value then
        FPS.enabled = true
        if not FPS.gui then buildFPSGui() end
        if FPS.gui then FPS.gui.Enabled = true end
        FPS.frames, FPS.elapsed = 0, 0

        if not FPS.conn then
            FPS.conn = RunService.RenderStepped:Connect(function(dt)
                if not FPS.enabled then return end
                FPS.frames = FPS.frames + 1
                FPS.elapsed = FPS.elapsed + dt
                if FPS.elapsed >= 1 then
                    local fps = math.floor(FPS.frames / FPS.elapsed + 0.5)
                    FPS.frames, FPS.elapsed = 0, 0
                    local ping = "--"
                    pcall(function()
                        local network = Stats:FindFirstChild("Network")
                        local serverStats = network and network:FindFirstChild("ServerStatsItem")
                        local dataPing = serverStats and serverStats:FindFirstChild("Data Ping")
                        if dataPing then ping = math.floor(dataPing:GetValue() + 0.5) end
                    end)
                    if FPS.label then
                        FPS.label.Text = "FPS: " .. fps .. " | PING: " .. ping .. " ms"
                    end
                end
            end)
        end
    else
        FPS.enabled = false
        if FPS.conn then
            pcall(function() FPS.conn:Disconnect() end)
            FPS.conn = nil
        end
        if FPS.gui then FPS.gui.Enabled = false end
    end
end

local IsFixingGenerator = false
local GenLastEnd = -math.huge
local GEN_END_COOLDOWN = 2

local function isPlayersNear(distance)
    local character = LocalPlayer.Character
    local myRoot = character and character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return false end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if root and (root.Position - myRoot.Position).Magnitude < distance then
                return true
            end
        end
    end
    return false
end

local function puzzleOpen()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    return (playerGui and playerGui:FindFirstChild("PuzzleUI")) ~= nil
end

task.spawn(function()
    while not destroyed do
        task.wait(0.2)
        pcall(function()
            if not Config.GenEnabled or IsFixingGenerator then return end
            if os.clock() - GenLastEnd < GEN_END_COOLDOWN then return end
            if not puzzleOpen() then return end

            local character = LocalPlayer.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local gameMap = getGameMap()
            if not (root and gameMap) then return end

            for _, generator in ipairs(gameMap:GetChildren()) do
                if generator:IsA("Model") and generator.Name == "Generator"
                    and generator:FindFirstChild("Main") then
                    if (generator.Main.Position - root.Position).Magnitude < 6.7 then
                        IsFixingGenerator = true
                        task.spawn(function()
                            pcall(function()
                                local remotes = generator:FindFirstChild("Remotes")
                                local progress = generator:FindFirstChild("Progress")
                                local remoteEvent = remotes and remotes:FindFirstChildOfClass("RemoteEvent")
                                if not (remoteEvent and progress) then return end

                                local lastProgress = progress.Value
                                local stalled = 0

                                while Config.GenEnabled
                                    and generator.Parent
                                    and progress.Value < 100
                                    and puzzleOpen() do
                                    local base = (not isPlayersNear(32) and Config.GenSpeedUp and 1.5)
                                        or Config.GenCooldown
                                    task.wait(base + (math.random() * 0.2 - 0.1))

                                    if not Config.GenEnabled or not generator.Parent then break end
                                    if progress.Value >= 100 then break end
                                    if not puzzleOpen() then break end

                                    pcall(function() remoteEvent:FireServer() end)

                                    if progress.Value > lastProgress then
                                        lastProgress = progress.Value
                                        stalled = 0
                                    else
                                        stalled = stalled + 1
                                        if stalled >= 8 then break end
                                    end
                                end
                            end)
                            GenLastEnd = os.clock()
                            IsFixingGenerator = false
                        end)
                        break
                    end
                end
            end
        end)
    end
end)

task.spawn(function()
    if type(hookfunction) ~= "function" then return end
    local assets = ReplicatedStorage:FindFirstChild("Assets")
    local killers = assets and assets:FindFirstChild("Killers")
    local azure = killers and (killers:FindFirstChild("Azure") or killers:FindFirstChild("!Azure"))
    local qteModule = azure and azure:FindFirstChild("cl_ConstructQTE", true)
    if not qteModule then return end

    local ok, module = pcall(require, qteModule)
    if not ok or type(module) ~= "table" or type(module.new) ~= "function" then return end

    local originalNew
    originalNew = hookfunction(module.new, function(self, ...)
        local instance = originalNew(self, ...)
        task.delay(0.05, function()
            if destroyed then return end
            if Config.GenDisarm and type(instance) == "table"
                and type(instance.AddProgress) == "function" then
                pcall(function() instance:AddProgress(100) end)
            end
        end)
        return instance
    end)
end)

local function applyJohnDoeFixes()
    local inGame = getInGame()
    if not inGame then return end

    for _, child in ipairs(inGame:GetChildren()) do
        if Config.JDTrails and child.Name:find("JohnDoeTrail") then
            for _, part in ipairs(child:GetChildren()) do
                if part:IsA("BasePart") then part.CanTouch = false end
            end
        end
        if Config.JDFootprints and child.Name:find("Shadows") then
            for _, part in ipairs(child:GetChildren()) do
                if part:IsA("BasePart") then part.CanTouch = false end
            end
        end
        if Config.JDSpikes and child.Name == "SpikeCollision" and child:IsA("BasePart") then
            child.Size = Vector3.new(11, 3.5, 3.5)
            child.Shape = Enum.PartType.Cylinder
        end
    end
end

task.spawn(function()
    while not destroyed do
        task.wait(0.5)
        if Config.JDTrails or Config.JDFootprints or Config.JDSpikes then
            pcall(applyJohnDoeFixes)
        end
    end
end)

local SprintModule = nil

task.spawn(function()
    local ok, module = pcall(function()
        return require(ReplicatedStorage:WaitForChild("Systems", 5)
            :WaitForChild("Character", 5)
            :WaitForChild("Game", 5)
            :WaitForChild("Sprinting", 5))
    end)
    if ok and module then
        SprintModule = module
    end
end)

task.spawn(function()
    while not destroyed do
        task.wait(0.5)
        if SprintModule and Config.StamPreset ~= "Original" then
            pcall(function()
                if Config.StamPreset == "Infinite" then
                    rawset(SprintModule, "Stamina", SprintModule.MaxStamina or math.huge)
                elseif SprintModule.Stamina and SprintModule.MaxStamina
                    and SprintModule.Stamina < SprintModule.MaxStamina * 0.8 then
                    local rate = Config.StamPreset == "Semi-Realistic" and 0.005 or 0.0025
                    rawset(SprintModule, "Stamina",
                        math.min(SprintModule.Stamina + SprintModule.MaxStamina * rate,
                            SprintModule.MaxStamina))
                end
            end)
        end
    end
end)

task.spawn(function()
    while not destroyed do
        task.wait(0.5)
        pcall(function()
            if Config.NoliNPC then
                local playersFolder = Workspace:FindFirstChild("Players")
                local killersFolder = playersFolder and playersFolder:FindFirstChild("Killers")
                if killersFolder then
                    for _, killer in ipairs(killersFolder:GetChildren()) do
                        if killer.Name:lower() == "noli" and not Players:GetPlayerFromCharacter(killer) then
                            killer.Parent = Lighting
                            pcall(function()
                                killer:PivotTo(killer:GetPivot() * CFrame.new(0, -100, 0))
                            end)
                        end
                    end
                end
                local themes = Workspace:FindFirstChild("Themes")
                if themes then
                    for _, sound in ipairs(themes:GetChildren()) do
                        if sound:IsA("Sound") and sound.Name:find("FakeLayer") then
                            sound:Destroy()
                        end
                    end
                end
            end

            if Config.NoliTimer then
                local timer = ReplicatedStorage:FindFirstChild("RoundTimer")
                if timer then
                    local timeLeft = timer:GetAttribute("TimeLeft")
                    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
                    local roundTimer = playerGui and playerGui:FindFirstChild("RoundTimer")
                    local ui = roundTimer and roundTimer:FindFirstChild("Time", true)
                    if timeLeft and ui then
                        pcall(function()
                            ui.Text = string.format("%d:%02d",
                                math.floor(timeLeft / 60), math.floor(timeLeft % 60))
                        end)
                    end
                end
            end
        end)
    end
end)

local function build(fn)

    if not ok then
        table.insert(uiErrors, tostring(err))
        warnf("UI 元素创建失败: %s", tostring(err))

local connections = {}

connections[#connections + 1] = RunService.RenderStepped:Connect(function(dt)
    if destroyed then return end
    for _, ring in pairs(rings) do
        ring:Update(dt)
    end
    pcall(updateFaceCones)
end)

connections[#connections + 1] = RunService.RenderStepped:Connect(function()
    if destroyed then return end
    pcall(punchAimTick)
end)

pcall(setFPSPing, Config.FPSPing)

task.spawn(function()
    while not destroyed do
        saveConfig(false)
        task.wait(3)
    end
end)

-- 运行时新加的 Sound / Animator（攻击音效常常是临时生成的）
connections[#connections + 1] = Workspace.DescendantAdded:Connect(function(instance)
    if destroyed then return end

    pcall(function()
        if instance:IsA("Sound") then
            local id = soundIdOf(instance)
            if id and SOUND_BLOCK_IDS[id] then
                watchSound(instance)
            end
        elseif instance:IsA("Animator") then
            local model = findOwnerModel(instance)
            if model then
                hookAnimator(model, instance)
            end
        end
    end)
end)

pcall(scanDetection)
pcall(updateESP)

-- SoundService 里的全局音效也要纳入监听
pcall(function()
    for _, descendant in ipairs(game:GetService("SoundService"):GetDescendants()) do
        if descendant:IsA("Sound") then
            watchSound(descendant)
        end
    end
end)

-- workspace 里预放好的 Sounds 文件夹（只递归 Folder，避免遍历整张地图）
pcall(function()
    local function scanSoundFolders(node, depth)
        if depth > 3 then return end
        for _, child in ipairs(node:GetChildren()) do
            if child:IsA("Sound") then
                watchSound(child)
            elseif child:IsA("Folder") or child:IsA("Configuration") then
                scanSoundFolders(child, depth + 1)
            end
        end
    end
    scanSoundFolders(Workspace, 0)
end)

task.spawn(function()
    while not destroyed do
        pcall(scanDetection)
        task.wait(0.4)
    end
end)

task.spawn(function()
    while not destroyed do
        pcall(updateESP)
        task.wait(0.2)
    end
end)

local lastStatus = ""

task.spawn(function()
    while not destroyed do
        local targets = getTargetsInRange(Config.Radius)
        inRange = #targets

        if statusTag then
            local text = string.format("%d studs · %d in range",
                math.floor(Config.Radius + 0.5), inRange)
            if text ~= lastStatus then
                lastStatus = text
                pcall(function()
                    statusTag:SetTitle(text)
                end)
            end
        end

        task.wait(0.3)
    end
end)

function TURN.Block(target, reason)
    -- target 省略时直接发一次 Block；给出目标时仍要求目标在白圈内。
    if target then
        return triggerBlock(target, reason or "api")
    end
    return fireBlockAbility()
end

function TURN.GetTargets()
    return getTargetsInRange(Config.Radius)
end

function TURN.TestBlock()
    return fireBlockAbility()
end

function TURN.IsArmed()
    return Config.AutoBlock
end

TURN.Window = Window
TURN.Tab = Tab
TURN.Rings = rings
TURN.ESP = ESP
TURN.UIErrors = uiErrors

function TURN.Destroy()
    if destroyed then return end
    destroyed = true

    for _, connection in ipairs(connections) do
        pcall(function() connection:Disconnect() end)
    end
    connections = {}

    pcall(saveConfig, true)
    pcall(stopHDT)
    if hdtScanConn then
        pcall(function() hdtScanConn:Disconnect() end)
    end
    pcall(restoreAimState)
    pcall(function() setFPSPing(false) end)

    local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
    local fpsGui = playerGui and playerGui:FindFirstChild("TURN_v1_FPSPing")
    if fpsGui then fpsGui:Destroy() end

    pcall(clearESP)

    for model, ring in pairs(rings) do
        pcall(function() ring:Destroy() end)
        rings[model] = nil
    end

    for model, cone in pairs(faceCones) do
        for _, part in ipairs(cone.Lines) do
            pcall(function() part:Destroy() end)
        end
        for _, part in ipairs(cone.Tips) do
            pcall(function() part:Destroy() end)
        end
        faceCones[model] = nil
    end

    pcall(function() Window:Destroy() end)

    if getgenv and getgenv().TURNopensource == TURN then
        getgenv().TURNopensource = nil
    end
end

if getgenv then
